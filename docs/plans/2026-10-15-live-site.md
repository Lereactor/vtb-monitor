# Live Site and Analytics Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** A public web site that shows the VTB detectors live, an incident feed filtered by
service, analytics with a plain-language summary, and a password-protected technical section.

**Architecture:** The VPS run that already checks the detectors also writes every check and
event to SQLite, computes the analytics, and force-pushes `public.json` + `private.enc` to the
`data` branch of `Lereactor/vtb-monitor-live`. The static React site (GitHub Pages, `main`
branch of the same repo) reads those files from raw.githubusercontent.com; the password
decrypts `private.enc` in the browser (same PBKDF2 → AES-GCM scheme as the agents site).
Nothing in this path may block alerts.

**Tech Stack:** Python 3.11/3.12 stdlib `sqlite3`, `cryptography` (AES-GCM, already on the
VPS as 41.0.7), git over an SSH deploy key; React 18 + Vite 5 + TypeScript, hand-written SVG
charts, Vitest.

Design: `docs/plans/2026-10-15-live-site-design.md`.

**Conventions for every task**
- Run tests with `python -m pytest -q` from the repo root (Windows: plain `python`).
- Write Python files with the Write/Edit tools, not bash heredocs (heredocs mangle `\n`).
- Times are stored in UTC ISO (`2026-10-15T06:10:00+00:00`) and shown in MSK.
- Commit after each task, message ends with the Co-Authored-By line.

**Step 3: Implement** `monitor/history.py`:

```python
"""Every check and event, in SQLite, for the live site. Callers must not let errors here block alerts."""
import re
import sqlite3
from datetime import timedelta, timezone

COUNT_RE = re.compile(r"жалоб за час:\s*(\d+)")
KEEP_CHECKS_DAYS = 90  # older checks are rolled up into hourly totals
SCHEMA = """
CREATE TABLE IF NOT EXISTS checks (ts TEXT, service TEXT, source TEXT, status TEXT,
                                   count INTEGER, details TEXT);
CREATE INDEX IF NOT EXISTS checks_ts ON checks (ts);
CREATE TABLE IF NOT EXISTS hourly (hour TEXT, service TEXT, source TEXT, checks INTEGER,
                                   outages INTEGER, errors INTEGER, max_count INTEGER,
                                   PRIMARY KEY (hour, service, source));
CREATE TABLE IF NOT EXISTS events (ts TEXT, kind TEXT, service TEXT, source TEXT, text TEXT);
CREATE INDEX IF NOT EXISTS events_ts ON events (ts);
CREATE TABLE IF NOT EXISTS posts (id INTEGER PRIMARY KEY, ts TEXT, kind TEXT, text TEXT, url TEXT);
"""


def connect(path):
    db = sqlite3.connect(path)
    db.executescript(SCHEMA)
    return db


def iso(moment):
    return moment.astimezone(timezone.utc).isoformat(timespec="seconds")


def count_from_details(details):
    match = COUNT_RE.search(details or "")
    return int(match.group(1)) if match else None


def record_run(db, now, results, events):
    ts = iso(now)
    db.executemany("INSERT INTO checks VALUES (?, ?, ?, ?, ?, ?)",
                   [(ts, service_id, r.source, r.status, count_from_details(r.details), r.details)
                    for service_id, service_results in results.items() for r in service_results])
    db.executemany("INSERT INTO events VALUES (?, ?, ?, ?, ?)",
                   [(ts, e.kind, e.service, e.source, e.text) for e in events])
    db.commit()


def roll_up(db, now):
    """Whole hours older than KEEP_CHECKS_DAYS move from checks to hourly."""
    cutoff = iso((now - timedelta(days=KEEP_CHECKS_DAYS)).replace(minute=0, second=0, microsecond=0))
    db.execute("""INSERT OR REPLACE INTO hourly
                  SELECT substr(ts, 1, 13), service, source, count(*), sum(status = 'OUTAGE'),
                         sum(status = 'SOURCE_ERROR'), max(count)
                  FROM checks WHERE ts < ? GROUP BY 1, 2, 3""", (cutoff,))
    db.execute("DELETE FROM checks WHERE ts < ?", (cutoff,))
    db.commit()
```

```python
"""One-off: rebuild history from journal lines written before the history existed.

ssh vps 'journalctl -u vtb-monitor.service -o short-iso --no-pager' > journal.txt
python -m monitor.backfill --db history.db journal.txt
"""
import argparse
import re
from datetime import datetime

from . import history
from . import state as st
from .sources.base import SourceResult

LINE_RE = re.compile(r"^(\S+) \S+ python3\[(\d+)\]: \[(\w+)/(\w+)\] (OK|OUTAGE|SOURCE_ERROR|DISABLED) ?(.*)$")


def parse_journal(lines):
    """[(run time, {service: [SourceResult]})], one run per process id, in time order."""
    runs = {}
    for line in lines:
        match = LINE_RE.match(line.strip())
        if not match:
            continue
        ts, pid, service, source, status, details = match.groups()
        when, results = runs.setdefault(pid, (datetime.fromisoformat(ts), {}))
        results.setdefault(service, []).append(SourceResult(source, status, details))
    return sorted(runs.values(), key=lambda run: run[0])


def backfill(db, lines):
    """Replays runs older than the first live check through the alert state machine."""
    first_live = db.execute("SELECT min(ts) FROM checks").fetchone()[0]
    state = st.new_state()
    done = 0
    for when, results in parse_journal(lines):
        if first_live and history.iso(when) >= first_live:
            break
        events = []
        for service_id, service_results in results.items():
            st.process(state, service_id, service_results, when, events)
        history.record_run(db, when, results, events)
        done += 1
    return done


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--db", required=True)
    parser.add_argument("journal")
    args = parser.parse_args()
    with open(args.journal, encoding="utf-8") as f:
        print(f"backfilled runs: {backfill(history.connect(args.db), f.read().splitlines())}")


if __name__ == "__main__":
    main()
```

```python
"""Official posts: the GitHub channel run appends them to posts.json, the VPS imports them."""
import json
from pathlib import Path

import requests

from .history import iso

KEEP = 300


def append(path, new_posts, now):
    """ts is when the post was found (within ~5 min of publication)."""
    path = Path(path)
    data = json.loads(path.read_text(encoding="utf-8")) if path.exists() else []
    data += [{"id": p.id, "ts": iso(now), "kind": p.kind, "text": p.text, "url": p.url} for p in new_posts]
    path.write_text(json.dumps(data[-KEEP:], ensure_ascii=False, indent=1) + "\n", encoding="utf-8")


def import_from_github(db, token, repo, session=requests):
    resp = session.get(f"https://api.github.com/repos/{repo}/contents/posts.json",
                       headers={"Authorization": f"Bearer {token}",
                                "Accept": "application/vnd.github.raw+json"}, timeout=15)
    if resp.status_code == 404:
        return 0
    resp.raise_for_status()
    rows = [(p["id"], p["ts"], p["kind"], p["text"], p["url"]) for p in json.loads(resp.text)]
    db.executemany("INSERT OR REPLACE INTO posts VALUES (?, ?, ?, ?, ?)", rows)
    db.commit()
    return len(rows)
```

```python
@dataclass
class Incident:
    service: str
    start: datetime
    end: datetime | None          # None while still open
    sources: list                 # in the order they joined
    peak: int | None = None       # max complaints/hour among count sources during it
    post: dict | None = None      # matching official post

    def minutes(self, now):
        return int(((self.end or now) - self.start).total_seconds() // 60)
```

### Task 7: Period analytics and the summary text

**Files:** Modify `monitor/analytics.py`; Test `tests/test_analytics.py`.

`PERIODS = {"24h": 1, "7d": 7, "30d": 30, "all": None}`, `SERVICE_FILTERS = ("all", "vtb", "invest")`.

`report(db, now)` → `{f"{period}|{service}": block}` where a block is:

```python
{
  "kpi": {"incidents": 2, "minutes": 35, "uptime": 0.9965,   # share of the period without an incident
          "confirmed": 1, "lead_min": 12},                    # median minutes from start to bank post
  "per_day": [{"date": "2026-10-14", "incidents": 1, "minutes": 20}],    # MSK dates, every day of the period
  "complaints": {"vtb/sboyrf": [["2026-10-15T05:00", 12], …]},          # hourly max count per count source
  "heatmap": [[…24 numbers…] × 7],      # mean complaints/hour by MSK weekday (Mon=0) × hour, all count sources summed
  "monitors": [{"service": "vtb", "source": "detector404", "first": 3, "solo": 1, "alarms": 4,
                "down_episodes": 0, "error_minutes": 15}],
  "summary": "За 7 дней: 2 сбоя ВТБ, суммарно 35 мин; …",
}
```

Rules:
- An incident belongs to a period if it overlaps it; minutes are clipped to the period.
- `uptime = 1 - incident_minutes / period_minutes` per service; for `all` services the mean
  of the two; `all` period starts at the first check.
- `first` = incidents where the source is `sources[0]`; `solo` = incidents with one source;
  `alarms` = incidents the source is in; `down_episodes` = `source_down` events;
  `error_minutes` = 5 × SOURCE_ERROR checks.
- `complaints`/`heatmap` read `checks` for the last 90 days and `hourly.max_count` beyond.
- Summary (no LLM), built from the block, ВТБ and Инвестиции named separately:
  - no incidents: «За 7 дней сбоев не было. Больше всего жалоб — по пятницам около 19:00.»
  - otherwise: «За 7 дней: 2 сбоя ВТБ (35 мин), 0 — Мои Инвестиции. Самый длинный — вт 14.10
    в 15:25, 20 мин. Банк подтвердил 1 из 2, в среднем через 12 мин после нашего оповещения.
    Первым обычно срабатывает DETECTOR404. Одиночных срабатываний: 1. DownRadar не отвечал 3 раза.»
  Use correct Russian plurals via a helper `plural(n, "сбой", "сбоя", "сбоев")`.

Tests: one test per rule above with a small hand-built history, plus `plural` for 1, 2, 5, 11,

```python
import json
import os

from cryptography.hazmat.primitives.ciphers.aead import AESGCM

ITERATIONS = 600_000


def derive_key(password, salt, iterations=ITERATIONS):
    return hashlib.pbkdf2_hmac("sha256", password.encode(), salt, iterations, 32)


def encrypt_json(payload, key, salt, iterations=ITERATIONS):
    iv = os.urandom(12)
    ct = AESGCM(key).encrypt(iv, json.dumps(payload, ensure_ascii=False).encode(), None)
    b64 = lambda raw: base64.b64encode(raw).decode()
    return {"v": 1, "iter": iterations, "salt": b64(salt), "iv": b64(iv), "ct": b64(ct)}


def decrypt_json(blob, password):
    key = derive_key(password, base64.b64decode(blob["salt"]), blob["iter"])
    plain = AESGCM(key).decrypt(base64.b64decode(blob["iv"]), base64.b64decode(blob["ct"]), None)
    return json.loads(plain)
```

### Task 9: Publish to the `data` branch

**Files:** Create `monitor/publish.py`; Modify `monitor/main.py:_after_run`; Test `tests/test_publish.py`.

```python
"""Force-push public.json + private.enc as a single commit to the site repo's data branch.

Not GitHub Pages (≈10 builds/hour limit): the site reads raw.githubusercontent.com.
"""
import json
import os
import shutil
import subprocess
from pathlib import Path


def publish(work_dir, remote, public, private_blob, ssh_key=None):
    work = Path(work_dir)
    shutil.rmtree(work, ignore_errors=True)  # one fresh commit each time: no history to grow
    work.mkdir(parents=True)
    (work / "public.json").write_text(json.dumps(public, ensure_ascii=False), encoding="utf-8")
    (work / "private.enc").write_text(json.dumps(private_blob), encoding="utf-8")
    env = {**os.environ}
    if ssh_key:
        env["GIT_SSH_COMMAND"] = (f"ssh -i {ssh_key} -o IdentitiesOnly=yes "
                                  f"-o UserKnownHostsFile={Path(ssh_key).parent / 'known_hosts'} "
                                  "-o StrictHostKeyChecking=accept-new")
    git = lambda *args: subprocess.run(["git", "-C", str(work), *args], check=True, env=env,
                                       capture_output=True, text=True, timeout=60)
    git("init", "-q", "-b", "data")
    git("add", "-A")
    git("-c", "user.name=vtb-monitor", "-c", "user.email=vtb-monitor@localhost",
        "commit", "-q", "-m", "data")
    git("push", "-q", "-f", remote, "data")
```

Test with a local bare repo as `remote` (`git init --bare`), then
`git --git-dir=<bare> show data:public.json` equals the payload; a second publish leaves

`main._after_run`: if `SITE_REMOTE`, `SITE_KEY`, `SITE_SALT` are in the environment, build
